*) Phi = Phi_matrix(p::ZZMPolyRingElem)
Given an integer polynomial p of degree two in two variables, returns the Phi-matrix associated to it.

*) qM = qM_matrix(Phi::AbstractAlgebra.Generic.MatSpaceElem, u::ZZMatrix)
Given a Phi-matrix for HM(X,u), calculates the matrix q*M.

###############
#EXAMPLES
###############

In a Julia REPL type

include("path_to/HorrocksMumford-symmetries.jl")

# u-Matrix of the Horrocks-Mumford bundle
u = matrix(ZZ,5,5,[0 1 -1 -1 1;1 0 1 -1 -1;-1 1 0 1 -1;-1 -1 1 0 1;1 -1 -1 1 0])

S,(i,j) = polynomial_ring(ZZ, ["i","j"])
# Up to the action of the Heisenberg group all relevant polynomials are
Phi_polys = [ c1*i^2+c2*i*j+c3*j^2 for c1 in 0:4  for c2 in 1:4 for c3 in 0:4 ];^2

p = Phi_polys[1]

Phi = Phi_matrix(p)

# check that this matrix is the one of (47)
Phi/Phi[1,1]
Phi[1,1]^2

qM = qM_matrix(Phi,u)

# check that we obtain indeed the entries as claimed in Section 6.4
e = gens(base_ring(Phi))[1]
e1 = 5*(e^3+e^2+1)
e2 = e1/5+1
z = gens(base_ring(qM))
qM[1,1] == sum( z[mod(nu-1,1:5)]*z[mod(nu,1:5)]*z[mod(nu+1,1:5)]*(z[mod(nu-1,1:5)]*z[mod(nu+1,1:5)]-e2*z[mod(nu,1:5)]^2)  for nu in 0:4 ) + e1*prod(z)
qM[1,2] == sum( z[mod(nu-2,1:5)]*z[mod(nu,1:5)]*z[mod(nu+2,1:5)]*(e2*z[mod(nu-2,1:5)]*z[mod(nu+2,1:5)]-z[mod(nu,1:5)]^2)  for nu in 0:4 ) - e1*prod(z)
qM[2,1] == sum( z[mod(nu-1,1:5)]*z[mod(nu,1:5)]*z[mod(nu+1,1:5)]*(e2*z[mod(nu-1,1:5)]*z[mod(nu+1,1:5)]-z[mod(nu,1:5)]^2)  for nu in 0:4 ) - e1*prod(z)
qM[2,2] == sum( z[mod(nu-2,1:5)]*z[mod(nu,1:5)]*z[mod(nu+2,1:5)]*(z[mod(nu-2,1:5)]*z[mod(nu+2,1:5)]-e2*z[mod(nu,1:5)]^2)  for nu in 0:4 ) + e1*prod(z)
